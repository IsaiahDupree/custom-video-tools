30def create_sample_audio(output_path, duration=30): 

31 """ 

32 Create a simple sample audio file for testing if none is available 

33  

34 Args: 

35 output_path: Path to save the generated audio 

36 duration: Duration in seconds 

37  

38 Returns: 

39 Path to the created audio file 

40 """ 

41 try: 

42 # Generate a simple sine wave audio file 

43 cmd = [ 

44 'ffmpeg', '-y', 

45 '-f', 'lavfi', 

46 '-i', f'sine=frequency=440:duration={duration}', 

47 '-c:a', 'mp3', 

48 '-b:a', '192k', 

49 output_path 

50 ] 

51 

52 subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE) 

53 

54 if os.path.exists(output_path): 

55 logger.info(f"Created sample audio file: {output_path}") 

56 return output_path 

57 else: 

58 logger.error("Failed to create sample audio") 

59 return None 

60 except Exception as e: 

61 logger.error(f"Error creating sample audio: {e}") 

62 return None

64def add_background_music(video_path, music_path, output_path, speech_emphasis=1.5): 

65 """ 

66 Add background music to a video with volume optimization 

67  

68 Args: 

69 video_path: Path to input video 

70 music_path: Path to music track 

71 output_path: Path for output video 

72 speech_emphasis: Factor to emphasize speech over music 

73  

74 Returns: 

75 Path to output video 

76 """ 

77 try: 

78 # Volume optimization parameters 

79 audio_volume_factor = 0.25 * speech_emphasis # Lower value = quieter music 

80 

81 # Command to extract original audio 

82 temp_audio = os.path.join(os.path.dirname(output_path), "temp_original_audio.aac") 

83 extract_audio_cmd = [ 

84 'ffmpeg', '-y', '-i', video_path, '-vn', '-acodec', 'copy', temp_audio 

85 ] 

86 

87 # Run extraction 

88 subprocess.run(extract_audio_cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE) 

89 

90 # Command to add background music with volume control 

91 add_music_cmd = [ 

92 'ffmpeg', '-y', 

93 '-i', video_path, # Original video 

94 '-i', music_path, # Background music 

95 '-i', temp_audio, # Original audio 

96 '-filter_complex', 

97 f'[1:a]aloop=loop=-1:size=2e+09,volume={audio_volume_factor}[background];' + 

98 '[2:a]volume=1.0[speech];' + 

99 '[background][speech]amix=inputs=2:duration=first', 

100 '-c:v', 'copy', # Copy video stream 

101 '-c:a', 'aac', '-b:a', '192k', # Good quality audio 

102 output_path 

103 ] 

104 

105 # Run command to add music 

106 logger.info("Adding background music with optimized volume...") 

107 logger.info(f"Command: {' '.join(add_music_cmd)}") 

108 result = subprocess.run(add_music_cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True) 

109 if result.returncode != 0: 

110 logger.error(f"ffmpeg error: {result.stderr}") 

111 else: 

112 logger.info("ffmpeg command executed successfully") 

113 

114 # Clean up temp files 

115 if os.path.exists(temp_audio): 

116 os.remove(temp_audio) 

117 

118 if os.path.exists(output_path): 

119 logger.info(f"Background music added successfully: {output_path}") 

120 return output_path 

121 else: 

122 logger.error("Failed to add background music") 

123 return None 

124 

125 except Exception as e: 

126 logger.error(f"Error adding background music: {e}") 

127 return None
