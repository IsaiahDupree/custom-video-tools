131 def _add_background_music(self, video: VideoFileClip, music_path: str, output_path: str) -> str: 

132 """ 

133 Add background music to a video at an optimized volume level. 

134  

135 Args: 

136 video: VideoFileClip object of the input video 

137 music_path: Path to the music file 

138 output_path: Path where the output video will be saved 

139  

140 Returns: 

141 Path to the output video (may change if file exists and overwrite is False) 

142 """ 

143 try: 

144 log.info(f"Adding music: {os.path.basename(music_path)}") 

145 

146 # Analyze original audio volume 

147 log.info(f"Analyzing audio volume for: {os.path.basename(video.filename)}") 

148 audio_array = video.audio.to_soundarray() 

149 

150 # Calculate optimal volume level based on original audio 

151 volume_db = calculate_optimal_volume_db( 

152 audio_array, 

153 self.target_ratio, 

154 self.speech_emphasis 

155 ) 

156 log.info(f"Using calculated volume level: {volume_db:.2f} dB") 

157 

158 # Convert from dB to volume factor 

159 # volume_factor = 10**(volume_db/20.0) 

160 # equivalent to 10^(db/20) since dB = 20*log10(ratio) 

161 volume_factor = np.power(10, volume_db / 20.0) 

162 

163 # Load music file 

164 music = AudioFileClip(music_path) 

165 

166 # Adjust music duration to match video 

167 video_duration = video.duration 

168 log.info(f"Trimming music to match video duration ({video_duration:.2f}s)") 

169 

170 # If music is shorter than video, loop it 

171 if music.duration < video_duration: 

172 # Calculate how many repeats we need 

173 repeats = int(np.ceil(video_duration / music.duration)) 

174 music = music.loop(repeats)

176 # Trim music to video duration 

177 music = music.subclip(0, video_duration) 

178 

179 # Set music volume 

180 log.info(f"Setting music volume to {volume_db:.2f} dB (factor: {volume_factor:.4f})") 

181 music = music.volumex(volume_factor)

183 # Mix with original audio 

184 log.info("Mixing music with original audio") 

185 final_audio = CompositeAudioClip([video.audio, music]) 

186
