64def calculate_md5(file_path: str) -> str: 

65 """Calculate MD5 hash of a file.""" 

66 hash_md5 = hashlib.md5() 

67 with open(file_path, "rb") as f: 

68 for chunk in iter(lambda: f.read(4096), b""): 

69 hash_md5.update(chunk) 

70 return hash_md5.hexdigest()

72def download_file(url: str, target_path: str) -> bool: 

73 """Download a file from URL to target path with progress bar.""" 

74 try: 

75 with Progress( 

76 TextColumn("[bold blue]{task.description}"), 

77 BarColumn(), 

78 DownloadColumn(), 

79 TransferSpeedColumn(), 

80 ) as progress: 

81 task = progress.add_task(f"Downloading {os.path.basename(target_path)}", total=None) 

82 

83 response = requests.get(url, stream=True) 

84 response.raise_for_status() 

85 

86 # Get content length if available 

87 total_size = int(response.headers.get('content-length', 0)) 

88 if total_size: 

89 progress.update(task, total=total_size) 

90 

91 # Download the file 

92 with open(target_path, 'wb') as f: 

93 for chunk in response.iter_content(chunk_size=8192): 

94 if chunk: 

95 f.write(chunk) 

96 progress.update(task, advance=len(chunk)) 

97 

98 return True 

99 except Exception as e: 

100 console.print(f"[bold red]Error downloading {url}: {e}[/bold red]") 

101 return False
